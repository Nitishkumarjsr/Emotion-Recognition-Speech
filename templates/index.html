import requests

url = "http://127.0.0.1:5002/api/predict"
with open("voice_recording.wav", "rb") as f:
    files = {"file": f}
    response = requests.post(url, files=files)

result = response.json()
print("Detected Emotion:", result["result"]["consensus"]["label"])
print("Confidence:", result["result"]["consensus"]["confidence"], "%")
print("Valence:", result["result"]["consensus"]["valence"])
print("Arousal:", result["result"]["consensus"]["arousal"])
